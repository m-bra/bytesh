int popenrcollect(const char *command, char *buffer, size_t buf_size) 
{
    FILE *fp;
    char combined_cmd[1024];
    char line[256];

    // Redirect stderr to stdout using "2>&1" so we capture everything
    snprintf(combined_cmd, sizeof(combined_cmd), "%s 2>&1", command);

//printf("DBG[C]\n");
    fp = popen(combined_cmd, "r");
    if (fp == NULL) {
        return -1;
    }

    int fpflags = fcntl(fileno(fp), F_GETFL, 0);
    fcntl(fileno(fp), F_SETFL, fpflags | O_NONBLOCK);

    // Initialize buffer as empty string
    buffer[0] = '\0';

    // Read output line by line and append to the buffer
//printf("DBG[D]\n");
//
////

struct pollfd pfd;
pfd.fd = fileno(fp);
pfd.events = POLLIN;

int runstatus = 0xFF;
rep blk

//printf("DBG[E.-1]\n");

        while (fgetsnonl(line, sizeof(line), fp) != NULL) {
//printf("DBG[E]\n");
           if (strlen(buffer) + strlen(line) < buf_size - 1) {
                strcat(buffer, line);
            } else {
                printf("%s at %s:%d\n", "Warning: Buffer Overflow Exception caught", __FILE__, __LINE__); 
	            break;
            }
        }

        int wstatus;
        iff ({int r = waitpid(-1, &wstatus, WNOHANG);
              if (r == -1 && errno != ECHILD) goto err;
              r;}) 
         && WIFEXITED(wstatus)
        thn {runstatus = WEXITSTATUS(wstatus); break;}
//printf("DBG[F]\n");

    end

    return pclose(fp);

err:
iff errno
thn perror(mf("%s:%d", __FILE__, __LINE__));
els printf("Error at %s:%d\n", __FILE__, __LINE__);
    return 1;

}
